# Przelew PKO → PEKAO
Invoke-RestMethod -Method POST -Uri "http://localhost:8080/api/sorbnet/payments" `
  -ContentType "application/json" `
  -Body '{"senderBankId":"PKO","receiverBankId":"PEKAO","amount":100000,"title":"Test RTGS"}'
  # Depozyt z NBP na PKO
Invoke-RestMethod -Method POST `
  -Uri "http://localhost:8080/api/sorbnet/accounts/PKO/deposit?amount=500000&sourceBankId=NBP"

  # Przelew który przekroczy limit (gridlock) - PKO ma debtLimit 2mln, balance 5mln
# Wysyłamy 7.5mln żeby zejść poniżej -2mln
Invoke-RestMethod -Method POST -Uri "http://localhost:8080/api/sorbnet/payments" `
  -ContentType "application/json" `
  -Body '{"senderBankId":"PKO","receiverBankId":"PEKAO","amount":7500000,"title":"Test Gridlock"}'

  # Doładuj PKO żeby wyszedł z gridlocka (potrzebuje min ~9.6M żeby pokryć 7.5M przy balance ~5.4M)
Invoke-RestMethod -Method POST `
  -Uri "http://localhost:8080/api/sorbnet/accounts/PKO/deposit?amount=5000000&sourceBankId=NBP"

# Poczekaj max 30 sekund (tyle wynosi gridlock.interval-ms) i sprawdź status płatności
Start-Sleep -Seconds 35

# Sprawdź czy przelew zmienił status na SETTLED
Invoke-RestMethod -Uri "http://localhost:8080/api/sorbnet/accounts/PKO/status"
